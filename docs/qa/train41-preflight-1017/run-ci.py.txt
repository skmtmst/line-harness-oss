import sys
sys.path.insert(0, '/Users/kentakenta/lh-work/tools/hq/ci-local')
import runner
original_env = runner.Runner.base_env
def capped_env(self, job, work, temp):
    env = original_env(self, job, work, temp)
    env.update(VITEST_MAX_FORKS='4', VITEST_MAX_THREADS='4')
    return env
runner.Runner.base_env = capped_env
original_commands = runner.Runner.commands
def capped_commands(self, script, work, temp, env, log, label, *args, **kwargs):
    env = env.copy()
    if 'pnpm --filter @line-crm/db test' in script:
        env['VITEST_MAX_FORKS'] = '2'
    return original_commands(self, script, work, temp, env, log, label, *args, **kwargs)
runner.Runner.commands = capped_commands
sys.exit(runner.main())
